    if gunConnRem then gunConnRem:Disconnect() end

        if gunSoundActive and child:IsA("Tool") and child.Name == TARGET_TOOL_NAME then
            actionSound:Play()
        end
    end)

    gunConnRem = character.ChildRemoved:Connect(function(child)
        if gunSoundActive and child:IsA("Tool") and child.Name == TARGET_TOOL_NAME then
            actionSound:Play()

        end
    end)
end

local function applyGunAnim(enable)
    if enable then
        if L.Character then
            local h = L.Character:FindFirstChildOfClass("Humanoid")
            if h then
                if gabConn then gabConn:Disconnect() end
                gabConn = h.AnimationPlayed:Connect(function(t)
                    if not gunAnimActive then return end
                    local id = tostring(t.Animation.AnimationId):lower()
                    local nm = tostring(t.Name):lower()
                    if nm:find("throw") or nm:find("stab") or nm:find("slash") or id:find("throw") or id:find("247206411") then

                    if id:find("reload") or nm:find("reload") or nm:find("recarga") or id:find("3400611413") or id:find("3400610059") or t.Priority == Enum.AnimationPriority.Action then 
                        t:Stop()
                    end
                end)
            end
        end

        if gabConn then gabConn:Disconnect(); gabConn = nil end
    end
end

local function applyWorldPink(enable)
    if enable then
        originalLightingPink.Ambient = Lighting.Ambient
        originalLightingPink.OutdoorAmbient = Lighting.OutdoorAmbient
        originalLightingPink.FogColor = Lighting.FogColor

        Lighting.Ambient = Color3.fromRGB(245, 225, 230)
        Lighting.OutdoorAmbient = Color3.fromRGB(255, 215, 225)
        Lighting.FogColor = Color3.fromRGB(255, 230, 235)

        local colorCorrection = Lighting:FindFirstChild("WorldPinkCC") or Instance.new("ColorCorrectionEffect")
        colorCorrection.Name = "WorldPinkCC"
        colorCorrection.TintColor = Color3.fromRGB(255, 235, 240)
        colorCorrection.Brightness = 0.05
        colorCorrection.Saturation = -0.1
        colorCorrection.Parent = Lighting

        local PASTEL_PINK = Color3.fromRGB(255, 192, 203)
        local SOFT_WHITE = Color3.fromRGB(250, 250, 250)

        local function stylePart(object)
            if object:IsA("BasePart") and not object.Parent:FindFirstChildOfClass("Humanoid") then
                if not originalMapColors[object] then
                    originalMapColors[object] = object.Color
                end
                local name = object.Name:lower()
                if name:find("trim") or name:find("border") or name:find("detail") or name:find("frame") then
                    object.Color = SOFT_WHITE
                else
                    object.Color = PASTEL_PINK
                end
            end
        end

        for _, descendant in ipairs(W:GetDescendants()) do
            stylePart(descendant)
        end

        worldPinkConnection = W.DescendantAdded:Connect(stylePart)
    else
        if worldPinkConnection then
            worldPinkConnection:Disconnect()
            worldPinkConnection = nil
        end

        if originalLightingPink.Ambient then
            Lighting.Ambient = originalLightingPink.Ambient
            Lighting.OutdoorAmbient = originalLightingPink.OutdoorAmbient
            Lighting.FogColor = originalLightingPink.FogColor
        end

        local cc = Lighting:FindFirstChild("WorldPinkCC")
        if cc then cc:Destroy() end

        for part, color in pairs(originalMapColors) do
            if part and part.Parent then
                part.Color = color
            end
        end
        originalMapColors = {}
    end
end

-- SPEED ESCALA
local escalaConnections = {}
local SB, MS = 60, 280
local gReady, mActive, isClimb, cSpam = false, false, false, 0
local uT, cAD, uKey, isFast = 0, 0, "", false
local uAY, dAnt, cG, uTG = 0, 0, 0, 0
local camSp, tecSp = false, false
local isGlitching = false

local function frenaEscala()
    local c = L.Character
    if c and c:FindFirstChild("HumanoidRootPart") then
        local hrp = c.HumanoidRootPart
        local vY = hrp.Velocity.Y
        if math.abs(vY) > 350 then vY = 0 end
        hrp.Velocity = hrp.Velocity:Lerp(Vector3.new(0, vY, 0), 0.2)
    end
end

local function cleanEscala()
    gReady, mActive, cSpam, cAD, isFast, uKey = false, false, 0, 0, false, ""
    cG, uAY, dAnt, camSp, tecSp = 0, 0, 0, false, false
    isGlitching = false
    frenaEscala()
end

local function checkGlitchState(char)
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return false end
    local animator = hum:FindFirstChildOfClass("Animator")
    if animator then
        for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
            local name = string.lower(track.Name)
            local animId = track.Animation and track.Animation.AnimationId or ""
            if string.find(name, "swim") or string.find(animId, "swim") then
                return true
            end
        end
    end
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {char}
    params.FilterType = Enum.RaycastFilterType.Exclude
    local hitClose = W:Raycast(hrp.Position, hrp.CFrame.LookVector * 1.8, params)
    local hitDown = W:Raycast(hrp.Position, -Vector3.yAxis * 2.5, params)
    if hitClose and hitDown then return true end
    return hum:GetState() == Enum.HumanoidStateType.Swimming
end

local function stopSpeedEscala()
    cleanEscala()
    for _, cn in ipairs(escalaConnections) do
        if cn and cn.Connected then cn:Disconnect() end
    end
    escalaConnections = {}
end

local function startSpeedEscala()
    stopSpeedEscala()
    local function setupChar(char)
        cleanEscala()
        local hum = char:WaitForChild("Humanoid", 5)
        if not hum then return end
        local cState = hum.StateChanged:Connect(function(_, st)
            isClimb = (st == Enum.HumanoidStateType.Climbing)
            if isClimb then
                mActive = false
            elseif gReady and (st == Enum.HumanoidStateType.Jumping or st == Enum.HumanoidStateType.Freefall) then
                mActive = true
            end
        end)
        table.insert(escalaConnections, cState)
        local cAdd = char.ChildAdded:Connect(function(ch)
            if ch:IsA("Tool") and isClimb then
                cSpam = cSpam + 1
                if cSpam >= 3 then gReady = true end
            end
        end)
        table.insert(escalaConnections, cAdd)
        local cDel = char.ChildRemoved:Connect(function(ch)
            if ch:IsA("Tool") then
                task.defer(function()
                    if char and not char:FindFirstChildOfClass("Tool") and mActive then 
                        cleanEscala() 
                    end
                end)
            end
        end)
        table.insert(escalaConnections, cDel)
    end

    if L.Character then setupChar(L.Character) end
    table.insert(escalaConnections, L.CharacterAdded:Connect(setupChar))

    local cGlitchLoop = R.Heartbeat:Connect(function()
        local c = L.Character
        if c then isGlitching = checkGlitchState(c) end
    end)
    table.insert(escalaConnections, cGlitchLoop)

    local cJumpInp = U.JumpRequest:Connect(function()
        local c = L.Character
        if isGlitching or (c and checkGlitchState(c)) then
            if c and c:FindFirstChild("HumanoidRootPart") then
                c.HumanoidRootPart.Velocity = Vector3.new(c.HumanoidRootPart.Velocity.X, 140, c.HumanoidRootPart.Velocity.Z)
                isGlitching = false
            end
        end
    end)
    table.insert(escalaConnections, cJumpInp)

    local cInp = U.InputBegan:Connect(function(inp, gpe)
        if gpe or not mActive then return end
        if inp.KeyCode == Enum.KeyCode.A or inp.KeyCode == Enum.KeyCode.D then
            local actKey = inp.KeyCode.Name
            local t = os.clock()
            if (t - uT) > 0.3 then
                cAD = 1; tecSp = false
            else
                if actKey ~= uKey then
                    cAD = cAD + 1
                    if cAD >= 2 then tecSp = true end
                end
            end
            uT, uKey = t, actKey
        end
    end)
    table.insert(escalaConnections, cInp)

    local camera = W.CurrentCamera
    local cStep = R.RenderStepped:Connect(function()
        if isClimb or not mActive then return end
        local c = L.Character
        if not c or not c:FindFirstChild("HumanoidRootPart") or not c:FindFirstChild("Humanoid") then return end
        local hrp = c.HumanoidRootPart
        if hrp.Velocity.Magnitude > 320 then
            local v = hrp.Velocity
            local clampY = math.clamp(v.Y, -180, 180)
            local capXZ = Vector3.new(v.X, 0, v.Z)
            local maxSpeed = isFast and MS or SB
            if capXZ.Magnitude > maxSpeed then
                capXZ = capXZ.Unit * maxSpeed
            end
            hrp.Velocity = Vector3.new(capXZ.X, clampY, capXZ.Z)
        end

        local tAct = os.clock()
        if c.Humanoid.FloorMaterial ~= Enum.Material.Air or (tAct - uT > 0.15 and tAct - uTG > 0.15) then
            if isFast then frenaEscala() end
            isFast, cAD, tecSp, camSp, cG = false, 0, false, false, 0
        end

        if camera then
            local _, aY, _ = camera.CFrame:ToEulerAnglesYXZ()
            local dAng = aY - uAY
            if dAng > math.pi then dAng = dAng - (math.pi * 2)
            elseif dAng < -math.pi then dAng = dAng + (math.pi * 2) end
            if math.abs(dAng) > 0.08 then
                local dAct = math.sign(dAng)
                if dAct ~= dAnt then
                    if tAct - uTG < 0.25 then
                        cG = cG + 1
                        if cG >= 3 then camSp = true end
                    end
                    dAnt, uTG = dAct, tAct
                end
            end
            uAY = aY
        end

        if tecSp and camSp then isFast = true end
        if c.Humanoid.FloorMaterial == Enum.Material.Air then
            local md = c.Humanoid.MoveDirection
            if md.Magnitude > 0 then
                local vY = hrp.Velocity.Y
                local spd = isFast and MS or SB
                hrp.Velocity = hrp.Velocity:Lerp(Vector3.new(md.X * spd, vY, md.Z * spd), 0.18)
            else
                frenaEscala()
            end
        end
    end)
    table.insert(escalaConnections, cStep)
end

-- PLAYER CHAMS
local MM2_PLACE_ID = 142823291
local MMV_PLACE_ID = 384594589

local function updatePlayerAura(p, c)
    if not c then return end
    local h = c:FindFirstChild("YellowAura")
    if not h then
        h = Instance.new("Highlight")
        h.Name = "YellowAura"
        h.Adornee = c
        h.Parent = c
    end

    local placeId = game.PlaceId
    local isMM = (placeId == MM2_PLACE_ID or placeId == MMV_PLACE_ID)

    if isMM then
        local backpack = p:FindFirstChild("Backpack")
        local character = p.Character
        
        local hasKnife = (character and (character:FindFirstChild("Knife") or character:FindFirstChild("Asesino") or character:FindFirstChild("Murderer"))) or 
                         (backpack and (backpack:FindFirstChild("Knife") or backpack:FindFirstChild("Asesino") or backpack:FindFirstChild("Murderer")))
                         
        local hasGun = (character and (character:FindFirstChild("Gun") or character:FindFirstChild("Sheriff") or character:FindFirstChild("Pistola"))) or 
                       (backpack and (backpack:FindFirstChild("Gun") or backpack:FindFirstChild("Sheriff") or backpack:FindFirstChild("Pistola")))

        for _, obj in ipairs(p:GetChildren()) do
            if obj:IsA("BoolValue") or obj:IsA("StringValue") then
                local valLower = string.lower(obj.Name .. " " .. tostring(obj.Value))
                if valLower:find("knife") or valLower:find("murder") or valLower:find("asesino") then
                    hasKnife = true
                elseif valLower:find("gun") or valLower:find("sheriff") or valLower:find("pistola") then
                    hasGun = true
                end
            end
        end

        if hasKnife then
            h.FillColor = Color3.fromRGB(255, 0, 0)
            h.OutlineColor = Color3.fromRGB(255, 255, 255)
        elseif hasGun then
            h.FillColor = Color3.fromRGB(0, 0, 255)
            h.OutlineColor = Color3.fromRGB(255, 255, 255)
        else
            h.FillColor = Color3.fromRGB(255, 255, 255)
            h.OutlineColor = Color3.fromRGB(200, 200, 200)
        end
    else
        h.FillColor = Color3.fromRGB(255, 255, 255)
        h.OutlineColor = Color3.fromRGB(200, 200, 200)
    end
    h.FillTransparency = 0.5
end

local function applyAura(p)
    if p.Character then
        updatePlayerAura(p, p.Character)
    end
end

local function removeAura(c)
    if c and c:FindFirstChild("YellowAura") then c.YellowAura:Destroy() end
end

local function enableChams()
    for _, p in ipairs(P:GetPlayers()) do
        if p ~= L then
            applyAura(p)
            local conn1 = p.CharacterAdded:Connect(function(c)
                c:WaitForChild("HumanoidRootPart", 5)
                if chamsActive then applyAura(p) end
            end)
            table.insert(chamsConnections, conn1)
            
            local conn2 = R.Heartbeat:Connect(function()
                if chamsActive and p.Character then
                    updatePlayerAura(p, p.Character)
                end
            end)
            table.insert(chamsConnections, conn2)
        end
    end
end

local function disableChams()
    for _, conn in ipairs(chamsConnections) do conn:Disconnect() end
    chamsConnections = {}
    for _, p in ipairs(P:GetPlayers()) do if p.Character then removeAura(p.Character) end end
end

    if speedEscalaActive then startSpeedEscala() end
    if antiFlingActive then applyAntiFling(true) end
    if anticoinsActive then applyAntiCoins(true) end
end

    if SPEED_ENABLED and humanoid and root then
        local s = humanoid:GetState()
        if s == Enum.HumanoidStateType.Jumping or s == Enum.HumanoidStateType.Freefall then
            local d, v = humanoid.MoveDirection, root.AssemblyLinearVelocity
            root.AssemblyLinearVelocity = v:Lerp(Vector3.new(d.X * speedValue, v.Y, d.Z * speedValue), 0.25)
        end
    end
end)
